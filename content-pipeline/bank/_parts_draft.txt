def solution_parts(text):
    """The solution in its parts, in order. The teaching standard writes every
    solution the same way (standards/teaching.md): an Answer line, the
    principle, "How to tell:" with the working one step to a line, "Why not the
    others:" one line per option, and an Exam tip. Anything outside that shape
    is kept as a principle line rather than lost."""
    parts, section = [], 'principle'
    for block in re.split(r'\n\s*\n', (text or '').strip()):
        lines = [l.strip() for l in block.split('\n') if l.strip()]
        if not lines:
            continue
        head = lines[0]
        if re.match(r'answer\s*:', head, re.I):
            parts.append(('answer', None, re.sub(r'^answer\s*:\s*', '', head, flags=re.I)))
            lines, section = lines[1:], 'principle'
        elif re.match(r'how to tell\s*:', head, re.I):
            rest = re.sub(r'^how to tell\s*:\s*', '', head, flags=re.I)
            lines, section = ([rest] if rest else []) + lines[1:], 'step'
        elif re.match(r'why not the others\s*:', head, re.I):
            rest = re.sub(r'^why not the others\s*:\s*', '', head, flags=re.I)
            lines, section = ([rest] if rest else []) + lines[1:], 'why_not'
        elif re.match(r'exam tip\s*:', head, re.I):
            lines, section = [re.sub(r'^exam tip\s*:\s*', '', head, flags=re.I)] + lines[1:], 'tip'
        elif section in ('step', 'why_not'):
            section = 'principle'       # a paragraph after a list closes it
        for l in lines:
            m = re.match(r'^([A-E])[\.\)]\s+(.*)$', l) if section == 'why_not' else None
            parts.append((section, m.group(1) if m else None, m.group(2) if m else l))
    return parts
